# Python dependencies for the documentation site.
#
# Installed by .github/workflows/docs.yml with:
#   python -m pip install -r docs/requirements.txt
#
# Everything is pinned to an exact version. A docs build that silently changes
# behaviour because a transitive theme release shipped is worse than a build
# that fails loudly on an intentional bump, and `mkdocs build --strict` gives
# us no slack: a new extension default that emits a warning becomes a red CI
# run. Bump these through a pull request so CI proves the change.

mkdocs==1.6.1
mkdocs-material==9.6.14

# Required by mkdocs-material for the emoji/icon extensions it declares.
mkdocs-material-extensions==1.3.1

# admonition + details + superfences (incl. the mermaid custom fence) +
# highlight/inlinehilite/tabbed/tasklist all come from here.
pymdown-extensions==10.15

# Syntax highlighting engine behind pymdownx.highlight. Pinned explicitly
# because a Pygments release adding or renaming a lexer changes rendered output.
Pygments==2.19.1
